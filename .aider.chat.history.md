index.js

const express = require('express');
const cors = require('cors');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(cors());
app.use(express.json());

const todos = [
  {
    id: 1,
    title: 'Learn Express.js',
    completed: false
  },
  {
    id: 2,
    title: 'Build a To-Do REST API',
    completed: true
  }
];

app.get('/todos', (req, res) => {
  res.status(200).json(todos);
});

if (require.main === module) {
  app.listen(PORT, () => {
    console.log(`Server running on port ${PORT}`);
  });
}

module.exports = app;
```

# Express.js To-Do REST API

This is an Express.js To-Do REST API. It provides a simple backend for managing to-do items and currently exposes an endpoint to fetch all to-dos.

## Features

- Node.js and Express.js backend
- CORS enabled
- In-memory sample to-do data
- `GET /todos` endpoint that returns all to-dos as JSON

## Prerequisites

- Node.js installed
- npm installed

## Installation

```bash
npm install
```

## Running the server

```bash
npm start
```

The server runs on port `3000` by default. You can override it with the `PORT` environment variable:

```bash
PORT=4000 npm start
```

## API Endpoints

### GET /todos

Returns HTTP status `200` and a JSON list of all to-dos.

Example request:

```bash
curl http://localhost:3000/todos
```

Example response:

[

    "title": "Learn Express.js",
    "completed": false
  },
  {
    "id": 2,
    "title": "Build a To-Do REST API",
    "completed": true
  }
]
```
